Consultas prontas (Supabase → SQL Editor)

Funil dos últimos 30 dias
select * from public.funil_30_dias;
Contas novas por semana
select date_trunc('week', created_at)::date as semana, count(*) as contas
  from public.profiles
 group by 1 order by 1 desc limit 12;
Quem tem plano pago ou cortesia
select email, plan, plan_expires_at
  from public.profiles
 where plan <> 'free'
 order by plan_expires_at nulls last;
Últimos pagamentos
select p.created_at, pr.email, p.plan, p.status, p.amount, p.method
  from public.payments p
  join public.profiles pr on pr.id = p.user_id
 order by p.created_at desc limit 50;
Assinaturas mensais ativas
select pr.email, s.amount, s.next_payment_date
  from public.subscriptions s
  join public.profiles pr on pr.id = s.user_id
 where s.status = 'authorized'
 order by s.next_payment_date;
Liberar 90 dias de cortesia

Troque o e-mail. A pessoa precisa ter criado a conta antes.

update public.profiles
   set plan = 'trimestral', plan_expires_at = now() + interval '90 days', updated_at = now()
 where email = 'email-do-testador@exemplo.com';
E-mails automáticos enviados
select kind, count(*) as total, max(sent_at) as ultimo
  from public.email_log
 group by kind order by ultimo desc;
